import { MCPClient, type RequireToolApprovalContext } from "@mastra/mcp";

// Fallback for tools the server does not annotate: read tools by GitHub MCP naming convention.
const READ_TOOL_PREFIXES = ["get_", "list_", "search_"];
const READ_TOOLS = new Set(["issue_read", "pull_request_read", "actions_get", "actions_list"]);

/** True only for GitHub MCP tools known to be read-only (raw MCP name, e.g. "list_issues"). */
export function isGithubReadTool(toolName: string): boolean {
  return READ_TOOLS.has(toolName) || READ_TOOL_PREFIXES.some((prefix) => toolName.startsWith(prefix));
}

/**
 * Deny by default. The server's own `readOnlyHint` annotation decides when present (true skips
 * approval, false requires it); only when the server sends no hint do we fall back to the read
 * name allowlist above. Unknown, unannotated tools always need approval.
 */
export function githubToolNeedsApproval({
  toolName,
  annotations,
}: Pick<RequireToolApprovalContext, "toolName" | "annotations">): boolean {
  const hint = annotations?.readOnlyHint;
  if (typeof hint === "boolean") return !hint;
  return !isGithubReadTool(toolName);
}
